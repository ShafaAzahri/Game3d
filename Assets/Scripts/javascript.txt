async function renderIcon(IconComponent, color = "#000000", size = 256) {
  const svg = ReactDOMServer.renderToStaticMarkup(
    React.createElement(IconComponent, { color, size: String(size) })
  );
  const pngBuffer = await sharp(Buffer.from(svg)).png().toBuffer();
  return "image/png;base64," + pngBuffer.toString("base64");
}

async function makeSlide(pres, bgColor = C.bgSlide) {
  const slide = pres.addSlide();
  slide.background = { color: bgColor };
  return slide;
}

function addTitle(slide, text, y = 0.28, color = C.navy) {
  slide.addText(text, {
    x: 0.5, y: y, w: 9.0, h: 0.55,
    fontSize: 22, bold: true, color: color,
    fontFace: "Calibri", align: "left", margin: 0,
  });
  // Accent dot
  slide.addShape("rect", {
    x: 0.5, y: y + 0.62, w: 0.35, h: 0.07,
    fill: { color: C.teal }, line: { type: "none" }
  });
}

function addSubTitle(slide, text, x = 0.5, y = 1.0, w = 9.0, h = 0.35, color = C.slate) {
  slide.addText(text, {
    x, y, w, h,
    fontSize: 13, color: color,
    fontFace: "Calibri", align: "left", italic: true,
  });
}

function addSpeakerNotes(slide, notes) {
  slide.addNotes(notes);
}

function makeCodeBox(slide, code, x, y, w, h) {
  slide.addShape("rect", {
    x, y, w, h,
    fill: { color: C.codeBlock }, line: { type: "none" },
    rectRadius: 0.06
  });
  slide.addText(code, {
    x: x + 0.12, y: y + 0.1, w: w - 0.24, h: h - 0.2,
    fontSize: 9, fontFace: "Consolas", color: C.codeText,
    align: "left", valign: "top", margin: 0,
  });
}

function cardBox(slide, x, y, w, h, accentColor = C.teal) {
  // Shadow
  slide.addShape("rect", {
    x: x + 0.04, y: y + 0.04, w, h,
    fill: { color: "CBD5E1", transparency: 50 }, line: { type: "none" }

  // White card
  slide.addShape("rect", {
    x, y, w, h,
    fill: { color: C.white }, line: { color: C.grayLight, pt: 1 }
  });
  // Left accent
  slide.addShape("rect", {
    x, y, w: 0.07, h,
    fill: { color: accentColor }, line: { type: "none" }
  });
}

const int NUM_SENSORS   = 5;

const float Kp = 0.5, Ki = 0.01, Kd = 0.2;
const int   SETPOINT    = 2000;   // Tengah 5 sensor
const int   DEADLINE_US = 10000;  // Deadline τ₂ = 10 ms
const int   BASE_SPEED  = 150;

float integral = 0, prevError = 0;
int   motorLeft = 0, motorRight = 0;

void computePID() {
  unsigned long t_start = micros();  // ← Stopwatch mulai

  // Hitung error posisi
  float error = (float)(SETPOINT - weightedPos);

  // Komponen PID
  float P = Kp * error;
  integral += error;
  // Anti-windup: batasi integral
  integral = constrain(integral, -5000, 5000);

  float D = Kd * (error - prevError);

  float correction = P + I + D;

  // Terapkan koreksi ke kecepatan motor

  motorRight = constrain(BASE_SPEED - correction, 0, 255);

  unsigned long duration = micros() - t_start;  // ← Stopwatch berhenti
  
  // Deadline enforcement
  if (duration > DEADLINE_US) {
    integral  = 0;  // Reset integrator saat overrun
    Serial.print("[MISS] PID: "); Serial.println(duration);
